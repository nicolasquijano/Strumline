# Strumline: guitar to MIDI. VST3 and a standalone application (the standalone shows a MIDI output selector because the processor produces MIDI).
juce_add_plugin(Strumline
  VERSION ${PROJECT_VERSION}
  COMPANY_NAME "Nico"
  PLUGIN_MANUFACTURER_CODE Nico
  PLUGIN_CODE Strm
  FORMATS VST3 Standalone
  PRODUCT_NAME "Strumline"
  PLUGIN_NAME "Strumline"
  ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/icon/icon-256.png"
  ICON_SMALL "${CMAKE_CURRENT_SOURCE_DIR}/icon/icon-64.png"
  IS_SYNTH FALSE
  NEEDS_MIDI_INPUT FALSE
  NEEDS_MIDI_OUTPUT TRUE
  IS_MIDI_EFFECT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE
  VST3_CATEGORIES Fx
  BUNDLE_ID com.nicolasquijano.strumline)

# The chord converter (Basic Pitch through ONNX Runtime). ONNX Runtime (MIT) is not in the repository: its headers and shared library are expected in
# STRUMLINE_ORT_DIR, outside the repository like the other big downloads. Get it with `cmake -P tools/fetch_onnxruntime.cmake <folder>` (any system), or put
# the files in by hand: include/ and, for this system, runtimes/<rid>/native/ (the layout of the NuGet package) or bin/ (Windows, the older layout of this
# machine). Without it everything is built as before, with the one-note converter only.
set(STRUMLINE_ORT_DIR "${CMAKE_SOURCE_DIR}/../../SAS-datasets/onnxruntime" CACHE PATH "Folder with ONNX Runtime (see tools/fetch_onnxruntime.cmake)")
set(_gm_model "${CMAKE_SOURCE_DIR}/engine/models/nmp.onnx")
set(_gm_poly OFF)

# The runtime identifier and the name of the shared library on this system.
if(WIN32)
  set(_ort_rid win-x64)
  set(_ort_lib_glob "onnxruntime*.dll")
  set(_ort_main "onnxruntime.dll")
elseif(APPLE)
  if(CMAKE_OSX_ARCHITECTURES MATCHES "x86_64" AND NOT CMAKE_OSX_ARCHITECTURES MATCHES "arm64")
    set(_ort_rid osx-x64)
  elseif(CMAKE_OSX_ARCHITECTURES MATCHES "arm64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(_ort_rid osx-arm64)
  else()
    set(_ort_rid osx-x64)
  endif()
  set(_ort_lib_glob "libonnxruntime*.dylib")
  set(_ort_main "libonnxruntime.dylib")
else()
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(_ort_rid linux-arm64)
  else()
    set(_ort_rid linux-x64)
  endif()
  set(_ort_lib_glob "libonnxruntime*.so")
  set(_ort_main "libonnxruntime.so")
endif()
set(_ort_libdir "")
foreach(_cand "${STRUMLINE_ORT_DIR}/runtimes/${_ort_rid}/native" "${STRUMLINE_ORT_DIR}/bin" "${STRUMLINE_ORT_DIR}/lib")
  if(EXISTS "${_cand}/${_ort_main}")
    set(_ort_libdir "${_cand}")
    break()
  endif()
endforeach()
if(EXISTS "${STRUMLINE_ORT_DIR}/include/onnxruntime_cxx_api.h" AND _ort_libdir AND EXISTS "${_gm_model}")
  set(_gm_poly ON)
  file(GLOB _ort_libs "${_ort_libdir}/${_ort_lib_glob}")
  # The model (225 KB) goes inside the plug-in as a C array.
  file(READ "${_gm_model}" _hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
  file(SIZE "${_gm_model}" _size)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gen/nmp_model.h.tmp"
    "#pragma once\n// Generated by CMake from engine/models/nmp.onnx (Basic Pitch, Apache 2.0).\nnamespace gm {\ninline constexpr unsigned char kNmpModel[] = {${_hex}};\ninline constexpr unsigned long long kNmpModelSize = ${_size};\n}\n")
  file(COPY_FILE "${CMAKE_CURRENT_BINARY_DIR}/gen/nmp_model.h.tmp" "${CMAKE_CURRENT_BINARY_DIR}/gen/nmp_model.h" ONLY_IF_DIFFERENT)
  message(STATUS "Strumline: chord converter ON (ONNX Runtime from ${_ort_libdir})")
else()
  message(STATUS "Strumline: chord converter OFF (no ONNX Runtime for ${_ort_rid} in ${STRUMLINE_ORT_DIR}); one-note converter only")
endif()

target_sources(Strumline PRIVATE StrumlineProcessor.cpp StrumlineEditor.cpp StrumlineProcessor.h StrumlineEditor.h I18n.cpp I18n.h I18nEs.cpp I18nPt.cpp I18nIt.cpp
  I18nZh.cpp I18nFr.cpp I18nDe.cpp I18nJa.cpp I18nRu.cpp I18nKo.cpp
  I18nZhTw.cpp I18nTr.cpp I18nPl.cpp I18nNl.cpp
  I18nSv.cpp I18nId.cpp I18nVi.cpp I18nUk.cpp I18nFi.cpp I18nCa.cpp I18nHi.cpp)
# The translations are UTF-8 source files.
target_compile_options(Strumline PUBLIC "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
target_include_directories(Strumline PRIVATE ${CMAKE_SOURCE_DIR}/engine)
target_compile_definitions(Strumline PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
target_link_libraries(Strumline PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
if(_gm_poly)
  target_include_directories(Strumline PUBLIC "${STRUMLINE_ORT_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/gen")
  target_compile_definitions(Strumline PUBLIC STRUMLINE_POLY=1)
  target_link_libraries(Strumline PUBLIC ${CMAKE_DL_LIBS})  # the library is opened by hand (dlopen on Linux and macOS)
endif()
# ONNX Runtime's shared library goes next to every binary that can run the chord converter (the plug-in finds it by the folder of its own module; inside a
# macOS bundle that is Contents/MacOS).
function(strumline_copy_ort target)
  if(_gm_poly)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_ort_libs} "$<TARGET_FILE_DIR:${target}>")
  endif()
endfunction()

# A picture of the editor, drawn without a window (the test script runs for six seconds, so the picture has notes and a chord):
#   build\plugin\strumline_shot.exe docs\interface.png
add_executable(strumline_shot shot.cpp)
target_include_directories(strumline_shot PRIVATE ${CMAKE_SOURCE_DIR}/engine ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(strumline_shot PRIVATE Strumline juce::juce_audio_utils juce::juce_recommended_config_flags)

# The behaviour without a host: strumline_check prints PASS or FAIL for the MIDI it sends, the synth, the switches and the all-notes-off.
add_executable(strumline_check check.cpp)
target_include_directories(strumline_check PRIVATE ${CMAKE_SOURCE_DIR}/engine ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(strumline_check PRIVATE Strumline juce::juce_audio_utils juce::juce_recommended_config_flags)

# The converter against GuitarSet recordings (see tools/extract_guitarset.py): strumline_eval <folder> [seconds] [sensitivity dB]
add_executable(strumline_eval eval.cpp)
target_include_directories(strumline_eval PRIVATE ${CMAKE_SOURCE_DIR}/engine)
target_link_libraries(strumline_eval PRIVATE Strumline juce::juce_audio_utils juce::juce_recommended_config_flags)

foreach(t Strumline_VST3 Strumline_Standalone strumline_shot strumline_check strumline_eval)
  strumline_copy_ort(${t})
endforeach()
